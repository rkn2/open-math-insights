NAEP: how do states compare to the national average?¶

Real data from data/public/naep-math-trends/ (see its SOURCE.md). Raw scale scores aren't very meaningful to compare across grades, so instead we look at each state relative to the national public-school average in the same year and grade — a state's "gap" — and how that gap has moved over time.

In [1]:
import pandas as pd
import matplotlib.pyplot as plt

naep = pd.read_csv("../../data/public/naep-math-trends/naep_math_scale_scores.csv")

national = (
    naep[naep["jurisdiction"] == "NP"]
    .set_index(["grade", "year"])["avg_scale_score"]
    .rename("national_avg")
)

states = naep[naep["jurisdiction"] != "NP"].copy()
states["national_avg"] = states.set_index(["grade", "year"]).index.map(national)
states["gap_vs_national"] = states["avg_scale_score"] - states["national_avg"]
states.head()

fig, axes = plt.subplots(1, 2, figsize=(12, 4.5), sharey=True)

for grade, ax in zip([4, 8], axes):
    subset = states[states["grade"] == grade]
    for juris, group in subset.groupby("jurisdiction"):
        group = group.sort_values("year")
        ax.plot(group["year"], group["gap_vs_national"], marker="o", label=juris)
    ax.axhline(0, color="black", linewidth=0.8, linestyle="--")
    ax.set_title(f"Grade {grade}: state score \u2212 national average")
    ax.set_xlabel("Year")
    ax.set_ylabel("Scale-score points vs. national")
    ax.legend(fontsize=8)
    ax.grid(alpha=0.3)

fig.tight_layout()
plt.show()

first_year, last_year = naep["year"].min(), naep["year"].max()

change = (
    naep[naep["year"].isin([first_year, last_year])]
    .pivot_table(index=["jurisdiction", "grade"], columns="year", values="avg_scale_score")
)
change["change"] = change[last_year] - change[first_year]
change.sort_values("change")
