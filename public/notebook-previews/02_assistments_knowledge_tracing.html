import pandas as pd
import matplotlib.pyplot as plt

DATA_DIR = "../../data/public"

# encoding_errors="replace": the raw CSV has a handful of malformed byte
# sequences deep in a free-text column (not one we use here) — replacing
# rather than raising keeps this real-world, occasionally-messy public
# dataset loadable without silently truncating rows.
# low_memory=False: works around a pandas C-parser bug where combining
# usecols + chunked low-memory reads + encoding_errors="replace" can raise
# an internal IndexError while building a (harmless) mixed-dtype warning.
df = pd.read_csv(
    f"{DATA_DIR}/assistments-2009-2010-skill-builder/skill_builder_data.csv",
    usecols=[
        "user_id", "correct", "attempt_count", "hint_count", "hint_total",
        "skill_name", "ms_first_response", "overlap_time",
    ],
    encoding="utf-8",
    encoding_errors="replace",
    low_memory=False,
)
print(f"{len(df):,} rows, {df['user_id'].nunique():,} unique students")

# Correctness rate by skill, for skills with enough attempts to be meaningful
by_skill = (
    df.dropna(subset=["skill_name"])
    .groupby("skill_name")
    .agg(attempts=("correct", "size"), correct_rate=("correct", "mean"))
    .query("attempts >= 200")
    .sort_values("correct_rate")
)

hardest = by_skill.head(15)
fig, ax = plt.subplots(figsize=(9, 6))
ax.barh(hardest.index, hardest["correct_rate"], color="#4f46e5")
ax.set_xlabel("Correct rate")
ax.set_title("15 hardest skills (lowest correctness rate, min. 200 attempts)")
ax.invert_yaxis()
fig.tight_layout()
plt.show()

# Does asking for a hint correlate with getting the problem right?
df["used_hint"] = df["hint_count"] > 0
hint_effect = df.groupby("used_hint")["correct"].mean()
print(hint_effect)

fig, ax = plt.subplots(figsize=(5, 4))
ax.bar(["No hint used", "Used a hint"], hint_effect.values, color=["#0d9488", "#f97316"])
ax.set_ylabel("Correct rate")
ax.set_title("Correctness vs. hint usage")
plt.show()
