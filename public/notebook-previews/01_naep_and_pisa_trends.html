fig, axes = plt.subplots(1, 2, figsize=(12, 4.5), sharey=False)

for grade, ax in zip([4, 8], axes):
    subset = naep[naep["grade"] == grade]
    for juris, group in subset.groupby("jurisdiction"):
        group = group.sort_values("year")
        ax.plot(group["year"], group["avg_scale_score"], marker="o", label=juris)
    ax.set_title(f"NAEP Grade {grade} average math scale score")
    ax.set_xlabel("Year")
    ax.set_ylabel("Average scale score")
    ax.legend(fontsize=8)
    ax.grid(alpha=0.3)

fig.tight_layout()
plt.show()

latest_year = pisa["year"].max()
top_countries = (
    pisa[pisa["year"] == latest_year]
    .sort_values("pisa_math_all_average", ascending=False)
    .head(8)["entity"]
    .tolist()
)

fig, ax = plt.subplots(figsize=(9, 5))
for country in top_countries:
    group = pisa[pisa["entity"] == country].sort_values("year")
    ax.plot(group["year"], group["pisa_math_all_average"], marker="o", label=country)

ax.set_title(f"PISA average math score \u2014 top 8 countries in {latest_year}")
ax.set_xlabel("Year")
ax.set_ylabel("Average PISA math score")
ax.legend(fontsize=8)
ax.grid(alpha=0.3)
plt.show()
